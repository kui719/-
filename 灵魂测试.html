// ===================== 逻辑部分 =====================

let current = 0;

let scores = { zhuge:0, qin:0, sushi:0, jingke:0, libai:0, yuefei:0, taoyuan:0 };

let answers = new Array(40).fill(null);


function startQuiz() {

  document.getElementById('home').style.display = 'none';

  document.getElementById('quiz').style.display = 'block';

  current = 0;

  scores = { zhuge:0, qin:0, sushi:0, jingke:0, libai:0, yuefei:0, taoyuan:0 };

  answers = new Array(40).fill(null);

  showQuestion();

}


function showQuestion() {

  const q = questions[current];

  document.getElementById('qText').textContent = q.q;

  document.getElementById('currentNum').textContent = current + 1;

  const percent = Math.round((current / 40) * 100);

  document.getElementById('progressPercent').textContent = percent + '%';

  document.getElementById('progressFill').style.width = percent + '%';


  const optionsEl = document.getElementById('options');

  optionsEl.innerHTML = '';

  

  q.options.forEach((opt, idx) => {

    const div = document.createElement('div');

    div.className = 'option' + (answers[current] === idx ? ' selected' : '');

    div.textContent = opt.t;

    div.onclick = () => selectOption(idx);

    optionsEl.appendChild(div);

  });


  document.getElementById('prevBtn').style.display = current > 0 ? 'block' : 'none';

  document.getElementById('nextBtn').disabled = answers[current] === null;

  document.getElementById('nextBtn').textContent = current === 39 ? '查看结果' : '下一题';

  

  // 重新触发动画

  const area = document.getElementById('questionArea');

  area.classList.remove('fade-in');

  void area.offsetWidth;

  area.classList.add('fade-in');

}


function selectOption(idx) {

  answers[current] = idx;

  document.querySelectorAll('.option').forEach((el, i) => {

    el.classList.toggle('selected', i === idx);

  });

  document.getElementById('nextBtn').disabled = false;

}


function nextQuestion() {

  if (answers[current] === null) return;

  

  if (current < 39) {

    current++;

    showQuestion();

  } else {

    calculateResult();

  }

}


function prevQuestion() {

  if (current > 0) {

    current--;

    showQuestion();

  }

}


function calculateResult() {

  // 重新计算分数（支持修改答案）

  scores = { zhuge:0, qin:0, sushi:0, jingke:0, libai:0, yuefei:0, taoyuan:0 };

  

  answers.forEach((ansIdx, qIdx) => {

    if (ansIdx === null) return;

    const s = questions[qIdx].options[ansIdx].s;

    for (let key in s) {

      scores[key] += s[key];

    }

  });


  // 找出最高分

  let maxScore = -1;

  let resultKey = 'zhuge';

  for (let key in scores) {

    if (scores[key] > maxScore) {

      maxScore = scores[key];

      resultKey = key;

    }

  }


  showResult(resultKey);

}


function showResult(key) {

  const c = characters[key];

  document.getElementById('quiz').style.display = 'none';

  document.getElementById('result').style.display = 'block';

  document.getElementById('result').classList.add('fade-in');


  document.getElementById('resName').textContent = c.name;

  document.getElementById('resCategory').textContent = c.category;

  document.getElementById('resSlogan').textContent = '「' + c.slogan + '」';

  

  const traitsEl = document.getElementById('resTraits');

  traitsEl.innerHTML = '';

  c.traits.forEach(t => {

    const span = document.createElement('span');

    span.className = 'trait';

    span.textContent = t;

    traitsEl.appendChild(span);

  });


  document.getElementById('resReason').textContent = c.reason;

  document.getElementById('resAdvantage').textContent = c.advantage;

  document.getElementById('resEaster').textContent = c.easter;

}


function restart() {

  document.getElementById('result').style.display = 'none';

  document.getElementById('home').style.display = 'block';

  document.getElementById('home').classList.add('fade-in');

}
